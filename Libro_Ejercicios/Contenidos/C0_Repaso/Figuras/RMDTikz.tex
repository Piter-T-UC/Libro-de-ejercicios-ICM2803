\tikzset{>=latex} % for LaTeX arrow head
\colorlet{veccol}{orange!90!black}
\colorlet{myblue}{blue!60!black}
\tikzstyle{vector}=[->,thick,veccol]
\def\Rad{1.4}
\def\rad{0.03}
\def\N{9}



% RADIAL OUTWARD
\begin{tikzpicture}
  \fill[myblue] (0,0) circle (\rad);
  \foreach \i [evaluate={\ang=\i*360/\N;}] in {0,...,\N}{
    \draw[vector] (\ang:0.1*\Rad) --++ (\ang:\Rad);
  }
  \node at (0,-1.35*\Rad) {$\div{{\color{veccol}\vb{v}}} > 0$};
\end{tikzpicture}


% RADIAL INWARD
\begin{tikzpicture}
  \fill[myblue] (0,0) circle (\rad);
  \foreach \i [evaluate={\ang=\i*360/\N;}] in {0,...,\N}{
    \draw[vector] (\ang:1.1*\Rad) -- (\ang:0.1*\Rad);
  }
  \node at (0,-1.35*\Rad) {$\nabla\cdot {{\color{veccol}\vec{v}}} < 0$};
\end{tikzpicture}
